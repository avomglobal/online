function doPost(e) { var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var d = JSON.parse(e.postData.contents); if (d.action) return adminApi(d); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName('Applications') || ss.insertSheet('Applications'); var email = String(d.email || '').trim().toLowerCase(); if (!email || email.indexOf('@') < 1) return jsonOut({ok: false, error: 'Please enter a valid email.'}); d.email = email; // Reject an email that has already applied if (sh.getLastRow() > 1) { var v = sh.getDataRange().getValues(), ei = v[0].indexOf('email'); for (var i = 1; i < v.length; i++) { if (String(v[i][ei]).trim().toLowerCase() === email) { return jsonOut({ok: false, duplicate: true, error: v[i][2] === 'Confirmed' ? 'This email is already registered. Sign in on the lessons page with your batch code.' : 'This email has already applied. Check your inbox and spam folder for the confirmation email, or contact the school.'}); } } } var keys = Object.keys(d), id = Utilities.getUuid(); if (sh.getLastRow() === 0) sh.appendRow(['Timestamp', 'ID', 'Status'].concat(keys)); sh.appendRow([new Date(), id, 'Pending'].concat(keys.map(function (k) { return d[k]; }))); MailApp.sendEmail(email, 'Action required: confirm your ' + SCHOOL_NAME + ' registration', 'Hello ' + d.name + ',\n\nThank you for applying. Please confirm your registration by opening this link:\n' + SCRIPT_URL + '?confirm=' + id + '\n\nOnce confirmed, you will receive your welcome email.'); return jsonOut({ok: true}); } finally { lock.releaseLock(); } }
