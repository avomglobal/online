Staff admin (admin.html) ---------- function adminApi(d) { var cache = CacheService.getScriptCache(), fails = Number(cache.get('fails') || 0); if (fails >= 5) return jsonOut({ok: false, error: 'Too many wrong attempts. Try again in 10 minutes.'}); var keys = (PropertiesService.getScriptProperties().getProperty('STAFF_KEYS') || '') .split(',').map(function (k) { return k.trim(); }).filter(String); if (!d.key || keys.indexOf(String(d.key)) < 0) { cache.put('fails', fails + 1, 600); return jsonOut({ok: false, error: 'Wrong staff key.'}); } var ss = SpreadsheetApp.getActiveSpreadsheet(), tz = Session.getScriptTimeZone(); var app = ss.getSheetByName('Applications'), st = ss.getSheetByName('Students'); var wanted = String(d.email || '').trim().toLowerCase(); if (d.action === 'list') { var out = {ok: true, apps: [], students: [], batches: []}; if (app && app.getLastRow() > 1) { var v = app.getDataRange().getValues(), h = v[0]; for (var i = 1; i < v.length; i++) { out.apps.push({ when: Utilities.formatDate(new Date(v[i][0]), tz, 'yyyy-MM-dd'), status: v[i][2], name: v[i][h.indexOf('name')], email: v[i][h.indexOf('email')], phone: v[i][h.indexOf('phone')], program: v[i][h.indexOf('program')] }); } } st.getDataRange().getValues().slice(1).forEach(function (r) { if (r[0]) out.students.push({email: r[0], batch: r[1]}); }); ss.getSheetByName('Batches').getDataRange().getValues().slice(1).forEach(function (r) { if (r[0]) out.batches.push({batch: r[0], code: r[1], start: r[2] ? Utilities.formatDate(new Date(r[2]), tz, 'yyyy-MM-dd') : ''}); }); return jsonOut(out); } if (d.action === 'resend') { var v2 = app.getDataRange().getValues(), ei = v2[0].indexOf('email'); for (var j = v2.length - 1; j >= 1; j--) { if (String(v2[j][ei]).trim().toLowerCase() === wanted && v2[j][2] === 'Pending') { MailApp.sendEmail(String(v2[j][ei]).trim(), 'Action required: confirm your ' + SCHOOL_NAME + ' registration', 'Please confirm your registration by opening this link:\n' + SCRIPT_URL + '?confirm=' + v2[j][1]); return jsonOut({ok: true}); } } return jsonOut({ok: false, error: 'No pending application found for that email.'}); } if (d.action === 'setBatch') { var s = st.getDataRange().getValues(); for (var k = 1; k < s.length; k++) { if (String(s[k][0]).trim().toLowerCase() === wanted) { st.getRange(k + 1, 2).setValue(d.batch); return jsonOut({ok: true}); } } return jsonOut({ok: false, error: 'Student not found.'}); } if (d.action === 'addBatch') { var p = String(d.start).split('-'); ss.getSheetByName('Batches').appendRow([d.batch, d.code, new Date(p[0], p[1] - 1, p[2])]); return jsonOut({ok: true}); } if (d.action === 'addLesson') { ss.getSheetByName('Lessons').appendRow([Number(d.lesson), d.title, d.videoId, d.notes || '']); return jsonOut({ok: true}); } return jsonOut({ok: false, error: 'Unknown action.'}); }
